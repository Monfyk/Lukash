cmake_minimum_required(VERSION 3.15)
project(LukashBoost CXX RC)
set(CMAKE_CXX_STANDARD 17)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded")
  add_compile_options(/utf-8 /W4)
endif()
add_executable(LukashBoost src/main.cpp src/app.rc)
target_link_libraries(LukashBoost ws2_32 iphlpapi advapi32)
if(MSVC)
  # маніфест уже вбудовано через app.rc
  target_link_options(LukashBoost PRIVATE /MANIFEST:NO)
else()
  target_link_options(LukashBoost PRIVATE -static -s)
endif()
